# main.py from fastapi import FastAPI from fastapi.responses import HTMLResponse from fastapi.staticfiles import StaticFiles from fastapi.middleware.cors import CORSMiddleware from pydantic import BaseModel from pathlib import Path import torch import torch.nn as nn import numpy as np import pandas as pd import joblib import math # ============================= # CONFIG # ============================= DEVICE = torch.device("cuda" if torch.cuda.is_available() else "cpu") STATE_DIM = 14 ACTION_DIM = 3 FERT_MIN = torch.tensor([0.0, 0.5, 0.4], device=DEVICE) FERT_MAX = torch.tensor([3.3, 4.4, 3.4], device=DEVICE) MODEL_PATH = Path(__file__).resolve().parent / "models" / "ppo_dap_checkjach_best.pt" # ============================= # PPO ACTOR CLASS # ============================= class PPOActor(nn.Module): def __init__(self, state_dim, action_dim, hidden_dim=256): super().__init__() self.net = nn.Sequential( nn.Linear(state_dim, hidden_dim), nn.ReLU(), nn.Linear(hidden_dim, hidden_dim), nn.ReLU(), ) self.mu = nn.Linear(hidden_dim, action_dim) self.log_std = nn.Parameter(torch.ones(action_dim) * -0.5) def forward(self, x): h = self.net(x) mu = self.mu(h) std = torch.exp(self.log_std) return mu, std # ============================= # LOAD PPO MODEL # ============================= actor = PPOActor(STATE_DIM, ACTION_DIM).to(DEVICE) if not MODEL_PATH.is_file(): raise FileNotFoundError(f"Checkpoint not found: {MODEL_PATH}") ckpt = torch.load(MODEL_PATH, map_location=DEVICE, weights_only=False) actor.load_state_dict(ckpt["actor_state_dict"]) actor.eval() # Load scaler used during PPO training if "scaler_X" in ckpt: scaler_X = ckpt["scaler_X"] else: raise KeyError("scaler_X not found in PPO checkpoint") # ============================= # LOAD MLR & RFR MODELS # ============================= mlr_pipeline = joblib.load("models/mlr_fertilizer_pipeline.pkl") rfr_pipeline = joblib.load("models/fert_recommendation_rf_pipeline.joblib") FEATURE_ORDER = [ "ph", "organic_matter", "total_nitrogen", "p2o5", "potassium", "boron", "zinc", "sand", "clay", "slit", "lat_sin", "lat_cos", "lon_sin", "lon_cos" ] # ============================= # FASTAPI APP # ============================= app = FastAPI(title="Fertilizer Recommendation API") app.add_middleware( CORSMiddleware, allow_origins=["*"], allow_credentials=True, allow_methods=["*"], allow_headers=["*"], ) # Serve static UI BASE_DIR = Path(__file__).resolve().parent app.mount("/static", StaticFiles(directory=BASE_DIR / "static"), name="static") @app.get("/", response_class=HTMLResponse) def serve_ui(): return open(BASE_DIR / "static" / "index.html", encoding="utf-8").read() # ============================= # INPUT SCHEMA # ============================= class SoilInput(BaseModel): lat: float lon: float ph: float organic_matter: float total_nitrogen: float p2o5: float potassium: float boron: float zinc: float sand: float clay: float slit: float # ============================= # HELPER FUNCTIONS # ============================= def encode_latlon(lat, lon): lat_rad = np.deg2rad(lat) lon_rad = np.deg2rad(lon) return np.sin(lat_rad), np.cos(lat_rad), np.sin(lon_rad), np.cos(lon_rad) def build_features(data: SoilInput): lat_sin, lat_cos, lon_sin, lon_cos = encode_latlon(data.lat, data.lon) features = [ data.ph, data.organic_matter, data.total_nitrogen, data.p2o5, data.potassium, data.boron, data.zinc, data.sand, data.clay, data.slit, lat_sin, lat_cos, lon_sin, lon_cos ] return torch.tensor([features], dtype=torch.float32, device=DEVICE), pd.DataFrame([{ "ph": data.ph, "organic_matter": data.organic_matter, "total_nitrogen": data.total_nitrogen, "p2o5": data.p2o5, "potassium": data.potassium, "boron": data.boron, "zinc": data.zinc, "sand": data.sand, "clay": data.clay, "slit": data.slit, "lat_sin": lat_sin, "lat_cos": lat_cos, "lon_sin": lon_sin, "lon_cos": lon_cos }])[FEATURE_ORDER] # ============================= # PREDICTION ENDPOINTS # ============================= @app.post("/predict/ppo") def predict_ppo(data: SoilInput): # Build raw features _, X_df = build_features(data) # 🔴 CRITICAL FIX: apply training scaler X_scaled = scaler_X.transform(X_df.values) # 🔴 convert to numpy X_tensor = torch.tensor( X_scaled, dtype=torch.float32, device=DEVICE ) with torch.no_grad(): mu, _ = actor(X_tensor) action_scaled = torch.tanh(mu) action_real = FERT_MIN + (action_scaled + 1) / 2 * (FERT_MAX - FERT_MIN) action_real = torch.clamp(action_real, FERT_MIN, FERT_MAX) return { "UREA": round(action_real[0, 0].item(), 2), "DAP": round(action_real[0, 1].item(), 2), "MOP": round(action_real[0, 2].item(), 2) } @app.post("/predict/mlr") def predict_mlr(data: SoilInput): _, X_df = build_features(data) preds = mlr_pipeline.predict(X_df)[0] return { "UREA": round(max(float(preds[0]), 0), 2), "DAP": round(max(float(preds[1]), 0), 2), "MOP": round(max(float(preds[2]), 0), 2) } @app.post("/predict/rfr") def predict_rfr(data: SoilInput): _, X_df = build_features(data) preds = rfr_pipeline.predict(X_df)[0] return { "UREA": round(max(float(preds[0]), 0), 2), "DAP": round(max(float(preds[1]), 0), 2), "MOP": round(max(float(preds[2]), 0), 2) } # 2. Nutrient Thresholding

This module analyzes clustered PPO predictions combined with disease outcomes to determine nutrient ranges that differentiate healthy and diseased plants. New soil samples can be compared against these thresholds to determine whether a nutrient is below, within, or above the healthy range.

Healthy N Range (Cluster 1): 3.10 – 4.25 kg/ropani
Healthy P Range (Cluster 1): 2.55 – 3.43 kg/ropani
Healthy K Range (Cluster 1): 2.66 – 3.65 kg/ropani
Disease Risk Zone (Cluster 1):
N < 3.10, P < 2.55, K < 2.66 kg/ropani

The entered soil values show that Nitrogen and Potassium are below their healthy cluster-specific Q10–Q90 ranges, while Phosphorus is within the optimal band. This indicates a nutrient deficiency pattern where low N and very low K increase the risk of reduced growth and potential disease susceptibility. Applying the recommended fertilizer doses, especially N and K, will help move the soil values closer to the healthy thresholds and reduce the risk zone.
